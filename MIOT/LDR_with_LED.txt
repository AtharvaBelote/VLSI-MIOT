int ldr = A0;
int led = 7;
int value;

void setup()
{
  pinMode(led, OUTPUT);
  Serial.begin(9600);
}

void loop()
{
  value = analogRead(ldr);

  Serial.println(value);

  if(value < 500)
  {
    digitalWrite(led, HIGH);
  }
  else
  {
    digitalWrite(led, LOW);
  }

  delay(500);
}
